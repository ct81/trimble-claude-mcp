            typeMessage = "textMessage"
            textMessageData = @{
                textMessage = "Hi, this is a PowerShell webhook test."
            }
        }
    } | ConvertTo-Json -Depth 10

    Write-Host "`nSending test request..." -ForegroundColor Cyan

    $response = Invoke-WebRequest `
        -UseBasicParsing `

        -Method POST `

        -ContentType "application/json" `
        -Body $body

    }

    if ($ptr -ne [IntPtr]::Zero) {
        [Runtime.InteropServices.Marshal]::ZeroFreeBSTR($ptr)
    }

    $token = $null
    $secureToken.Dispose()
