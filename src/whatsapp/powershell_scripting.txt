    $ApiToken = [Runtime.InteropServices.Marshal]::PtrToStringBSTR($Ptr)
}
finally {
    [Runtime.InteropServices.Marshal]::ZeroFreeBSTR($Ptr)
}

# Recipient WhatsApp number with country code, no + sign
# Example Singapore: 6591234567
$PhoneNumber = Read-Host "Enter recipient WhatsApp number (e.g. 6591234567)"

$Message = "Test message from PowerShell via GREEN-API. Time: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"

$ChatId = "${PhoneNumber}@c.us"

$Body = @{
    chatId  = $ChatId
    message = $Message
} | ConvertTo-Json -Depth 5

Write-Host "`nSending WhatsApp message..." -ForegroundColor Cyan
Write-Host "Recipient: $ChatId"
Write-Host "Message: $Message"

try {
    $Response = Invoke-RestMethod `
        -Uri $Url `
        -Method Post `
        -ContentType "application/json; charset=utf-8" `
        -Body ([System.Text.Encoding]::UTF8.GetBytes($Body)) `
        -TimeoutSec 30

    Write-Host "`nGREEN-API response:" -ForegroundColor Green
    $Response | ConvertTo-Json -Depth 5

    if ($Response.idMessage) {
        Write-Host "`nSUCCESS: GREEN-API accepted the send request." -ForegroundColor Green
        Write-Host "Message ID: $($Response.idMessage)"
    }
}
catch {
    Write-Host "`nFAILED to send message." -ForegroundColor Red
    Write-Host $_.Exception.Message

    if ($_.ErrorDetails.Message) {
        Write-Host $_.ErrorDetails.Message
    }
}
finally {
    $ApiToken = $null
    $SecureToken.Dispose()
}

# Trimble Connect MCP Status Check
# ==========================================

$BaseUrl = "https://trimble-claude-mcp-g86i.onrender.com"

Write-Host "`n==========================================" -ForegroundColor Cyan
Write-Host " TRIMBLE CONNECT MCP STATUS CHECK" -ForegroundColor Cyan
Write-Host "==========================================" -ForegroundColor Cyan

# 1. Check service health
Write-Host "`n[1] Checking service health..." -ForegroundColor Yellow

try {
    $health = Invoke-RestMethod `
        -Uri "$BaseUrl/health" `
        -Method Get `
        -TimeoutSec 30

    Write-Host "Health endpoint responded:" -ForegroundColor Green
    $health | ConvertTo-Json -Depth 10
}
catch {
    Write-Host "Health check failed: $($_.Exception.Message)" -ForegroundColor Red
}

# 2. Check MCP endpoint
Write-Host "`n[2] Checking MCP endpoint..." -ForegroundColor Yellow

$body = @{
    jsonrpc = "2.0"
    id      = 1
    method  = "initialize"
    params  = @{
        protocolVersion = "2025-11-25"
        capabilities    = @{}
        clientInfo      = @{
            name    = "powershell-status-check"
            version = "1.0.0"
        }
    }
} | ConvertTo-Json -Depth 10

try {
    $response = Invoke-WebRequest `
        -Uri "$BaseUrl/mcp" `
        -Method Post `
        -ContentType "application/json" `
        -Headers @{
            "Accept" = "application/json, text/event-stream"
        } `
        -Body $body `
        -TimeoutSec 30

    Write-Host "MCP HTTP Status: $([int]$response.StatusCode)" -ForegroundColor Green
    Write-Host "MCP Response:" -ForegroundColor Green
    Write-Host $response.Content

    if ($response.Headers["Mcp-Session-Id"]) {
        Write-Host "`nMCP Session ID returned." -ForegroundColor Green
    }
}
catch {
    Write-Host "MCP request failed." -ForegroundColor Red

    if ($_.Exception.Response) {
        Write-Host "HTTP Status: $([int]$_.Exception.Response.StatusCode)" -ForegroundColor Red
    }

    Write-Host $_.Exception.Message
}

Write-Host "`n==========================================" -ForegroundColor Cyan
Write-Host " CHECK COMPLETE" -ForegroundColor Cyan
Write-Host "==========================================" -ForegroundColor Cyan
